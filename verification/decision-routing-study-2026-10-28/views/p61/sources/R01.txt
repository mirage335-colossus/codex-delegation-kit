  33 RecordId Store::add(std::string_view text) {
  34     validate(text);
  35     if (records_.size() == capacity_) {
  36         throw std::length_error("collection is full");

  39         throw std::overflow_error("record identifiers exhausted");
  40     }
  41     const auto id = next_id_;

  45 }

  47 bool Store::update(RecordId id, std::string_view text) {
  48     validate(text);
  49     const auto found = std::find_if(records_.begin(), records_.end(),
  50         [id](const Record& row) { return row.id == id; });
  51     if (found == records_.end()) return false;

  53     found->text.swap(replacement);
  54     return true;
