```text
   1 // One outstanding poll is sufficient: transport order also orders task steps.
   2 export function startPolling(client, scheduler=globalThis, interval=25) {
   3   let stopped=false, timer=null;
   4   const schedule=()=>{if(!stopped)timer=scheduler.setTimeout(run,interval);};
   5   const run=()=>{
   6     timer=null;if(stopped||client.closed)return;
   7     if(client.failed){schedule();return;}
   8     client.send({type:'poll'}).catch(()=>{}).finally(schedule);
   9   };
  10   schedule();
  11   return ()=>{stopped=true;if(timer!==null)scheduler.clearTimeout(timer);timer=null;};
  12 }
```
