def remove_synthetic_directory(directory):
    if not directory.exists():
        return
    for path in (directory, *directory.rglob("*")):
        if path.is_dir():
            path.chmod(0o700)
    shutil.rmtree(directory)

    def snapshot_copy(self, name, *, current=False):
        snapshot = self.temp / name
        shutil.copytree(p.SNAPSHOT / "checkout", snapshot / "checkout")
        (snapshot / "manifest.json").write_bytes((p.SNAPSHOT / "manifest.json").read_bytes())
        if current:
            checkout = p.STUDY.parents[1]
            for relative in ("install.py", "agents/sol_researcher.toml"):
                shutil.copyfile(checkout / relative, snapshot / "checkout" / relative)
        self.refresh_snapshot_manifest(snapshot, 24 if current else 23)
        return snapshot

    def refresh_snapshot_manifest(self, snapshot, count):
        manifest = json.loads((snapshot / "manifest.json").read_text())
        manifest["runtime_inventory_entries"] = count
        root = snapshot / "checkout"
        manifest["copied_files_sha256"] = {
            str(path.relative_to(root)): p.digest(path) for path in root.rglob("*") if path.is_file()
        }
        (snapshot / "manifest.json").write_text(json.dumps(manifest))

    def test_rehashed_snapshot_inventory_mismatches_and_forgeries(self):
        cases = ("declared24-source23", "declared23-source24", "installer23-source24",
                 "installer24-source23", "unsupported-member", "duplicate-installer-member",
                 "wrong-installer-member", "unregistered-role")
        for case in cases:
            with self.subTest(case=case):
                old_source = case in ("declared24-source23", "installer24-source23")
                snapshot = self.snapshot_copy(case, current=not old_source)
                root = snapshot / "checkout"
                installer = root / "install.py"
                count = 23 if old_source else 24
                if case == "declared24-source23":
                    count = 24
                elif case == "declared23-source24":
                    count = 23
                elif case == "installer23-source24":
                    installer.write_bytes((p.SNAPSHOT / "checkout/install.py").read_bytes())
                elif case == "installer24-source23":
                    installer.write_bytes((p.STUDY.parents[1] / "install.py").read_bytes())
                elif case == "unsupported-member":
                    (root / "agents/sol_researcher.toml").rename(root / "agents/unsupported.toml")
                    installer.write_text(installer.read_text() + '\nFILES = FILES[:-1] + ("agents/unsupported.toml",)\n')
                elif case == "duplicate-installer-member":
                    installer.write_text(installer.read_text() + '\nFILES = FILES[:-1] + (FILES[0],)\n')
                elif case == "wrong-installer-member":
                    installer.write_text(installer.read_text() + '\nFILES = FILES[:-1] + ("agents/unsupported.toml",)\n')
                elif case == "unregistered-role":
                    installer.write_text(installer.read_text() + '\nROLES = dict(ROLES)\nROLES.pop("sol_researcher")\n')
                # Rehash the changed bytes: these must fail structural admission,
                # independently of stale-manifest hash rejection.
                self.refresh_snapshot_manifest(snapshot, count)
                with self.assertRaises(p.Hold):
                    p.frozen_inventory(snapshot)

    def test_claim_and_response_each_use_one_checked_snapshot(self):
        self.relay.enqueue(self.directory, self.request())

        reads = {outbox: 0, raw: 0}
        def replace_after_snapshot(path, maximum=2 * 1024 * 1024):
            data = original_snapshot(path, maximum)
            path = Path(path)
            if path in reads:
                reads[path] += 1
                if path == outbox:
                    changed = json.loads(data)
                    changed["request"]["parameters"]["search_query"][0]["q"] = "replacement after snapshot"
                    path.write_text(json.dumps(changed))
                else:
                    path.write_bytes(b"different unchecked response" * 100)
            return data
        with patch.object(self.relay, "read_snapshot", side_effect=replace_after_snapshot):
            frozen = self.relay.claim(self.directory, "one", "synthetic scope review")
            result = self.relay.publish(self.directory, "one", raw, status="returned", actual_operations=1,
                                        actual_calls=1, request_sha256=frozen["outbox_sha256"])
        self.assertEqual(reads, {outbox: 1, raw: 1})

        published = (self.directory / "relay/inbox/one/response.bin").read_bytes()
        self.assertEqual(published, expected_response)
        self.assertEqual(result["response_bytes"], len(expected_response))
        self.assertEqual(result["response_sha256"], __import__("hashlib").sha256(expected_response).hexdigest())
        self.assertEqual(json.loads((self.directory / "relay/control/published.json").read_text())["bytes"], len(expected_response))

    def test_close_intent_withholds_a_publication_already_holding_the_lock(self):

        intent_ready = threading.Event()
        errors, closer = [], []
        def close():
            try:
                self.relay.close_author(self.directory, {"reason": "synthetic first terminal"})
            except BaseException as error:
                errors.append(error)
        def interleave(path, value):
            original(path, value)
            if Path(path).name == "closing.json":
                intent_ready.set()
            elif Path(path).name == "receipt.json":
                thread = threading.Thread(target=close)
                closer.append(thread)
                thread.start()
                self.assertTrue(intent_ready.wait(1), "closure intent was not durable during publication")
        with patch.object(self.relay, "exclusive_json", side_effect=interleave):
            result = self.relay.publish(self.directory, "one", raw, status="returned", actual_operations=1,
                                        actual_calls=1, request_sha256=frozen["outbox_sha256"])
            for thread in closer:
                thread.join(timeout=2)
        self.assertFalse(errors)
        self.assertTrue(all(not thread.is_alive() for thread in closer))
        self.assertFalse(result["published_to_author"])
        self.assertFalse((self.directory / "relay/inbox/one").exists())
        self.assertTrue((self.directory / "relay/inbox/terminal.json").exists())

class ContainmentTests(unittest.TestCase):
    def setUp(self):
        self.temp = Path(tempfile.mkdtemp(prefix="primary-boundary-"))
        self.owned = self.temp / "owned"
        self.owned.mkdir()

    def tearDown(self):
        shutil.rmtree(self.temp)

    def boundary(self, command):
        return ["/usr/bin/bwrap", "--ro-bind", "/", "/", "--unshare-pid", "--unshare-ipc",
                "--die-with-parent", "--new-session", "--proc", "/proc", "--tmpfs", "/tmp",
                "--bind", str(self.owned), str(self.owned), "--ro-bind", str(self.temp / "denied"),
                str(self.temp / "denied"), "--", *command]

    def make_boundary(self):
        (self.temp / "denied").mkdir()

    def test_real_write_denial_private_proc_ipc_and_tmpfs(self):
        self.make_boundary()
        code = '''import os,pathlib
owned,denied=map(pathlib.Path,__import__('sys').argv[1:3])
(owned/'allowed').write_text('allowed')
try: (denied/'forbidden').write_text('forbidden')
except OSError: pass
else: raise AssertionError('outer write boundary failed')
pathlib.Path('/tmp/private-tmpfs').write_text('ephemeral')
assert os.readlink('/proc/self/ns/pid') != __import__('sys').argv[3]
assert os.readlink('/proc/self/ns/ipc') != __import__('sys').argv[4]
'''
        # Numeric PIDs can coincide; namespace identities establish separation.
        result = subprocess.run(self.boundary([sys.executable, "-B", "-c", code,
                                str(self.owned), str(self.temp / "denied"),
                                os.readlink('/proc/self/ns/pid'), os.readlink('/proc/self/ns/ipc')]),
                                capture_output=True, timeout=5)
        self.assertEqual(result.returncode, 0, result.stderr.decode())
        self.assertTrue((self.owned / "allowed").exists())
        self.assertFalse((self.temp / "denied/forbidden").exists())
        self.assertFalse(Path("/tmp/private-tmpfs").exists())

    def test_real_timeout_cleanup_includes_detached_grandchild(self):
        self.make_boundary()
        code = '''import os,time,pathlib
if os.fork()==0:
 os.setsid()
 if os.fork()==0:
  pathlib.Path(__import__('sys').argv[1]).write_text('ready')
  while True: time.sleep(.1)
 os._exit(0)
while True: time.sleep(.1)
'''
        owned = p.OwnedProcess(self.boundary([sys.executable, "-B", "-c", code, str(self.owned / "ready")]), dict(os.environ))
        deadline = time.monotonic() + 3
        while not (self.owned / "ready").exists() and time.monotonic() < deadline:
            owned.observe()
            time.sleep(.02)
        self.assertTrue((self.owned / "ready").exists())
        owned.observe()
        self.assertIsNotNone(owned.pid_namespace)
        release = owned.close(3)
        self.assertTrue(release["released"], release)
        self.assertGreaterEqual(len(release["observed_members"]), 2)

    def test_real_parent_death_releases_detached_grandchild(self):
        self.make_boundary()
        # A dedicated launcher is killed; no signal targets anything except it.
        child = "import os,time;os.fork();os.setsid() if os.getpid()!=1 else None;time.sleep(30)"
        launch = "import subprocess,time,json; p=subprocess.Popen(json.loads(__import__('sys').argv[1])); print(p.pid,flush=True);time.sleep(30)"
        parent = subprocess.Popen([sys.executable, "-B", "-c", launch, json.dumps(self.boundary([sys.executable, "-B", "-c", child]))],
                                  stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        launcher_pid = int(parent.stdout.readline())
        owned = object.__new__(p.OwnedProcess)
        owned.process = type("Process", (), {"pid": launcher_pid})()
        owned.pid_namespace, owned.members = None, {}
        for _ in range(20):
            owned.observe()
            if len(owned.members) >= 2:
                break
            time.sleep(.02)
        self.assertIsNotNone(owned.pid_namespace)
        parent.kill()
        parent.wait(timeout=2)
        deadline = time.monotonic() + 3
        while time.monotonic() < deadline:
            alive = [v for v in owned.members.values() if (now := p.identity(v["pid"])) and now["state"] != "Z"]
            if not alive:
                break
            time.sleep(.02)
        self.assertFalse(alive)
        parent.stdout.close()
        parent.stderr.close()
