def remove_synthetic_directory(directory):
    if not directory.exists():
        return
    for path in (directory, *directory.rglob("*")):
        if path.is_dir() and not path.is_symlink():
            path.chmod(0o700)
    shutil.rmtree(directory)


class NeutralProfileTests(unittest.TestCase):
    def setUp(self):
        self.temp = Path(tempfile.mkdtemp(prefix="neutral-unit-synthetic-"))
        self.addCleanup(remove_synthetic_directory, self.temp)
        self.neutral = p.neutral()

    def profile(self, task_name="research-task.md", **changes):
        value = {
            "schema": 1, "name": "neutral-v1", "candidate_label": "synthetic-exact-v9",
            "snapshot": str(SNAPSHOT), "source_manifest_sha256": SOURCE_SHA256,
            "project_manifest": str(PROJECT_MANIFEST),
            "project_manifest_sha256": p.digest(PROJECT_MANIFEST),
            "task_sha256": TASKS[task_name], "relay_services": {"public": True, "dependencies": True},
            "context": str(INPUTS / "application-context.md"), "context_sha256": CONTEXT_SHA256,
        }
        value.update(changes)
        path = self.temp / (task_name + ".json")
        path.write_text(json.dumps(value))
        return path, value

    def admit(self, profile, task_name="research-task.md", *, task=None, fixture=FIXTURE):
        return self.neutral.admit(profile, task or INPUTS / task_name, fixture, p.safe, p.digest)

    def synthetic_host(self):
        home = self.temp / "synthetic-home"
        ch = home / ".codex"
        ch.mkdir(parents=True)
        (ch / "config.toml").write_text('model = "gpt-6-astra"\n')
        (ch / "auth.json").write_text("synthetic sentinel; never read by driver")
        (ch / "state_5.sqlite").write_text("synthetic hidden history")
        (ch / "AGENTS.md").write_text("unmanaged-before\n" + p.BEGIN + "\nold\n" + p.END + "\nunmanaged-after\n")
        package = home / ".local/lib/node_modules/@openai/codex"
        binary = package / "bin/codex.js"
        native = package / "node_modules/@openai/codex-linux-x64/vendor/x86_64-unknown-linux-musl/bin/codex"
        for path in (binary, native):
            path.parent.mkdir(parents=True, exist_ok=True)
            path.write_text("synthetic nonexecutable input")
        return home, ch, binary

    def operation(self):
        directory = Path(tempfile.mkdtemp(prefix="neutral-prepared-synthetic-"))
        directory.rmdir()
        self.addCleanup(remove_synthetic_directory, directory)
        return directory

    def prepare(self, home, binary, task_name="research-task.md", **changes):
        profile, _ = self.profile(task_name, **changes)
        directory = self.operation()
        with patch.dict(os.environ, {"HOME": str(home)}, clear=True):
            prepared = p.prepare(directory, INPUTS / task_name, FIXTURE,
                                 profile=profile, home=home, codex=binary)
            self.assertEqual(p.verify_prepared(directory), prepared)
        return directory, prepared

    def test_exact_frozen_source_context_and_both_ordinary_tasks_admitted(self):
        self.assertEqual(self.neutral.NAME, "neutral-v1")
        self.assertEqual(self.neutral.SOURCE_MANIFEST_SHA256, SOURCE_SHA256)
        self.assertEqual(self.neutral.TASK_SHA256S, frozenset(TASKS.values()))
        self.assertEqual(self.neutral.CONTEXT_SHA256, CONTEXT_SHA256)
        self.assertEqual(p.digest(SNAPSHOT / "manifest.json"), SOURCE_SHA256)
        self.assertEqual(p.digest(INPUTS / "application-context.md"), CONTEXT_SHA256)
        identities = []
        for name, expected_hash in TASKS.items():
            with self.subTest(task=name):
                self.assertEqual(p.digest(INPUTS / name), expected_hash)
                profile, value = self.profile(name)
                admitted, identity = self.admit(profile, name)
                self.assertEqual(admitted, value)

    def test_exact_profile_shape_paths_and_enabled_services_required(self):
        profile, baseline = self.profile()
        cases = []
        for field in baseline:
            value = dict(baseline)
            value.pop(field)
            cases.append(("missing-" + field, value))
        for field in ("snapshot", "project_manifest", "context"):
            for bad in ("relative/path", None, 12):
                cases.append((field + "-" + repr(bad), dict(baseline, **{field: bad})))
        for name, changes in (
            ("unknown-profile", {"name": "neutral-v2"}),
            ("unfrozen-limit", {"limits": {"web_items": 2000}}),
            ("empty-label", {"candidate_label": " "}),
            ("invalid-schema", {"schema": 2}),
        ):
            cases.append((name, dict(baseline, **changes)))
        for services in ({"public": False, "dependencies": True}, {"public": True, "dependencies": False},
                         {"public": 1, "dependencies": True}, {"public": True},
                         {"public": True, "dependencies": True, "extra": True}, None):
            cases.append(("services-" + repr(services), dict(baseline, relay_services=services)))
        for label, value in cases:
            with self.subTest(case=label):
                profile.write_text(json.dumps(value))
                with self.assertRaises(ValueError):
                    self.admit(profile)

    def test_copy_time_mutations_fail_before_staged_installer_import(self):
        home, _, binary = self.synthetic_host()
        profile, _ = self.profile()
        original_copytree = shutil.copytree
        original_copyfile = shutil.copyfile
        original_write_bytes = Path.write_bytes
        frozen_before = {str(path.relative_to(SNAPSHOT)): p.digest(path)
                         for path in SNAPSHOT.rglob("*") if path.is_file()}
        for mutation in ("source-content", "source-extra", "source-missing", "fixture", "task", "context"):
            with self.subTest(mutation=mutation):
                directory = self.operation()
                mutated = []

                def change_copytree(source, destination, *args, **kwargs):
                    result = original_copytree(source, destination, *args, **kwargs)
                    destination = Path(destination)
                    if mutation.startswith("source-") and destination == directory / "source":
                        destination.chmod(0o700)
                        installer = destination / "install.py"
                        if mutation == "source-content":
                            installer.chmod(0o600)
                            original_write_bytes(installer, installer.read_bytes() + b"\n# Changed during staging.\n")
                        elif mutation == "source-extra":
                            original_write_bytes(destination / "unmanifested.py", b"# Synthetic extra member.\n")
                        else:
                            installer.unlink()
                        mutated.append(mutation)
                    elif mutation == "fixture" and destination == directory / "fixture":
                        target = destination / "AGENTS.md"
                        target.chmod(0o600)
                        original_write_bytes(target, target.read_bytes() + b"\n")
                        mutated.append(mutation)
                    return result

                def change_copyfile(source, destination, *args, **kwargs):
                    result = original_copyfile(source, destination, *args, **kwargs)
                    if mutation == "context" and Path(destination) == directory / "support/application-context.md":
                        target = Path(destination)
                        target.chmod(0o600)
                        original_write_bytes(target, target.read_bytes() + b"\n")
                        mutated.append(mutation)
                    return result

                def change_write_bytes(target, data):
                    if mutation == "task" and target == directory / "task.txt":
                        data += b"\n"
                        mutated.append(mutation)
                    return original_write_bytes(target, data)

                with patch.dict(os.environ, {"HOME": str(home)}, clear=True), \
                        patch.object(shutil, "copytree", side_effect=change_copytree), \
                        patch.object(shutil, "copyfile", side_effect=change_copyfile), \
                        patch.object(Path, "write_bytes", new=change_write_bytes), \
                        patch.object(p, "installer_at", side_effect=AssertionError("staged installer imported before input rejection")) as load_installer:
                    with self.assertRaises((p.Hold, ValueError)):
                        p.prepare(directory, INPUTS / "research-task.md", FIXTURE,
                                  profile=profile, home=home, codex=binary)
                self.assertEqual(mutated, [mutation])
                load_installer.assert_not_called()
                self.assertFalse((directory / "prepared.json").exists())
        self.assertEqual({str(path.relative_to(SNAPSHOT)): p.digest(path)
                          for path in SNAPSHOT.rglob("*") if path.is_file()}, frozen_before)

class NeutralRelayTests(unittest.TestCase):
    def setUp(self):
        self.temp = Path(tempfile.mkdtemp(prefix="neutral-relay-synthetic-"))
        self.addCleanup(remove_synthetic_directory, self.temp)
        self.neutral = p.neutral()
        spec = importlib.util.spec_from_file_location("neutral_test_scientific_relay", MODULE.with_name("scientific_relay.py"))
        self.relay = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(self.relay)

    def stage(self, name):
        directory = self.temp / name
        directory.mkdir()
        self.neutral.stage(directory, {"relay_services": {"public": True, "dependencies": True},
                                      "context": str(INPUTS / "application-context.md")},
                           {"revision": "synthetic-revision"}, 11, p.write_json)
        p.write_json(directory / "relay/inbox/dispatch.json", {"deadline_monotonic": time.monotonic() + 60})
        return directory

    @staticmethod
    def request(request_id, kind="web", parameters=None):
        return {"id": request_id, "author": "synthetic-root", "kind": kind,
                "parameters": {"search_query": [{"q": "synthetic offline request"}]} if parameters is None else parameters}

    def test_each_case_has_an_independent_web_meter_and_shared_helper_allocation(self):
        first, second = self.stage("opaque-a"), self.stage("opaque-b")
        first_request = self.request("same-stable-id", parameters={"search_query": [{"q": "one"}, {"q": "two"}]})
        self.relay.enqueue(first, first_request)
        self.assertFalse((second / "work/accounting/web/meter.json").exists())
        self.relay.enqueue(second, self.request("same-stable-id"))
        helper = dict(self.request("helper-id"), author="synthetic-child")
        self.relay.enqueue(first, helper)
        self.assertEqual(json.loads((first / "work/accounting/web/meter.json").read_text())["used"], {"author": 3})
        self.assertEqual(json.loads((second / "work/accounting/web/meter.json").read_text())["used"], {"author": 1})
        # Reserve synthetic local items only; no request is claimed or executed.

    def test_close_during_dependency_publication_withholds_receipt_and_retains_exact_audit(self):
        directory = self.stage("opaque")
        request = self.request("dependency-race", "dependency",
                               {"path": "docs/synthetic.md", "purpose": "synthetic ordinary dependency"})
        self.relay.enqueue(directory, request)
        claimed = self.relay.claim(directory, "dependency-race", "synthetic scope review")
        response = self.temp / "response.bin"
        response_bytes = b"Synthetic raw response with retained exact bytes.\n"
        response.write_bytes(response_bytes)
        dependency = self.temp / "dependency.bin"
        dependency_bytes = b"# Synthetic dependency\nNo external acquisition occurred.\n"
        dependency.write_bytes(dependency_bytes)
        inbox = directory / "relay/inbox/dependency-race"
        visible_dependency = directory / "relay/dependencies/dependency-race/docs/synthetic.md"
        closure_intent_durable = threading.Event()
        closure_started = threading.Event()
        closure_results, closure_errors = [], []
        original_sync = self.relay.sync_directory

        def close_author():
            try:
                closure_results.append(self.relay.close_author(directory, {"reason": "synthetic publication race"}))
            except BaseException as error:
                closure_errors.append(error)

        closer = threading.Thread(target=close_author, daemon=True)

        def sync_then_close(path):
            original_sync(path)
            if Path(path) == directory / "relay/inbox" and (directory / "relay/inbox/closing.json").exists():
                closure_intent_durable.set()
            if Path(path) == directory / "relay/dependencies" and not closure_started.is_set():
                closure_started.set()
                self.assertEqual(visible_dependency.read_bytes(), dependency_bytes)
                closer.start()
                self.assertTrue(closure_intent_durable.wait(1), "closure intent was not durable at publication barrier")
                self.assertFalse(inbox.exists(), "author receipt became visible before final closure recheck")

        try:
            with patch.object(self.relay, "sync_directory", side_effect=sync_then_close):
                result = self.relay.publish(directory, "dependency-race", response,
                                            status="returned", actual_operations=0, actual_calls=1,
                                            request_sha256=claimed["outbox_sha256"], dependency_file=dependency)
        finally:
            if closer.ident is not None:
                closer.join(timeout=3)
        self.assertTrue(closure_started.is_set())
        self.assertFalse(closer.is_alive(), "bounded closure thread did not finish after publisher released the lock")
        self.assertFalse(closure_errors)
        self.assertEqual(len(closure_results), 1)
        self.assertFalse(inbox.exists())
        self.assertFalse(visible_dependency.exists())
        self.assertFalse(result["published_to_author"])
        self.assertEqual(result["disposition"], "late_private_audit")
        window_path = directory / result["publication_window"]["audit_path"]
        self.assertTrue(window_path.is_relative_to(directory / "relay/control"))
        window = json.loads(window_path.read_text())
        self.assertTrue(window["dependency_temporarily_visible"])
        self.assertFalse(window["published_to_author"])
        self.assertEqual(window["disposition"], "late_private_audit")
        self.assertEqual(window["qualification"], result["publication_window"]["qualification"])
        timestamps = [window[key] for key in (
            "dependency_visibility_started_monotonic", "dependency_visibility_observed_monotonic",
            "final_live_check_started_monotonic", "final_live_check_completed_monotonic",
            "dependency_withdrawn_observed_monotonic")]
        self.assertEqual(timestamps, sorted(timestamps))
        self.assertLessEqual(closure_results[0]["closure_requested_monotonic"], timestamps[1])
        self.assertNotIn("response_rename_started_monotonic", window)
        audit = directory / result["audit_path"]
        retained_dependency = directory / result["dependency"]["path"]
        self.assertTrue(audit.is_relative_to(directory / "relay/control"))
        self.assertTrue(retained_dependency.is_relative_to(directory / "relay/control"))
        self.assertEqual((audit / "response.bin").read_bytes(), response_bytes)
        self.assertEqual(retained_dependency.read_bytes(), dependency_bytes)
        self.assertEqual(json.loads((audit / "receipt.json").read_text()), result)
        self.assertEqual(result["actual_tool_calls"], 1)
        state = json.loads((directory / "relay/control/published.json").read_text())
        self.assertEqual(state["bytes"], len(response_bytes) + len(dependency_bytes))
        self.assertEqual(state["requests"], ["dependency-race"])
        self.assertEqual(state["dispositions"]["dependency-race"]["delivery"], "late_private_audit")
        self.assertEqual(state["dispositions"]["dependency-race"]["actual_tool_calls"], 1)
        self.assertTrue((directory / "relay/inbox/terminal.json").exists())
        with self.assertRaises(ValueError):
            self.relay.publish(directory, "dependency-race", response,
                               status="returned", actual_operations=0, actual_calls=1,
                               request_sha256=claimed["outbox_sha256"], dependency_file=dependency)
