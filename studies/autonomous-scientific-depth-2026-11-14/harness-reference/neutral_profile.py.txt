def _module(name, filename):
    if name not in sys.modules:
        spec = importlib.util.spec_from_file_location(name, Path(__file__).with_name(filename))
        module = importlib.util.module_from_spec(spec)
        sys.modules[name] = module
        spec.loader.exec_module(module)
    return sys.modules[name]

def admit(path, task, fixture, safe, digest):
    profile = json.loads(safe(path).read_text())
    required = {"schema", "name", "candidate_label", "snapshot", "source_manifest_sha256",
                "project_manifest", "project_manifest_sha256", "task_sha256", "relay_services",
                "context", "context_sha256"}
    if set(profile) != required or type(profile["schema"]) is not int or profile["schema"] != 1 or profile["name"] != NAME:
        raise ValueError("unsupported neutral profile shape")
    if not isinstance(profile["candidate_label"], str) or not profile["candidate_label"].strip():
        raise ValueError("explicit candidate label required")
    for key in ("snapshot", "project_manifest", "context"):
        if not isinstance(profile[key], str) or not Path(profile[key]).is_absolute():
            raise ValueError("explicit absolute profile path required: " + key)
    snapshot, manifest_path, context = (safe(profile[key]) for key in ("snapshot", "project_manifest", "context"))
    if (profile["source_manifest_sha256"] != SOURCE_MANIFEST_SHA256
            or digest(snapshot / "manifest.json") != SOURCE_MANIFEST_SHA256):
        raise ValueError("exact neutral candidate manifest required")
    if digest(manifest_path) != profile["project_manifest_sha256"]:
        raise ValueError("project manifest changed")
    if profile["task_sha256"] not in TASK_SHA256S or digest(task) != profile["task_sha256"]:
        raise ValueError("exact declared neutral task required")
    if profile["context_sha256"] != CONTEXT_SHA256 or digest(context) != CONTEXT_SHA256:
        raise ValueError("exact common application context required")
    if (profile["relay_services"] != {"public": True, "dependencies": True}
            or any(type(value) is not bool for value in profile["relay_services"].values())):
        raise ValueError("both neutral services must be available")
    manifest = json.loads(manifest_path.read_text())
    rows = manifest["files"]
    expected = {row["path"]: {key: row[key] for key in ("sha256", "size_bytes")} for row in rows}
    actual = base.inventory(fixture, safe, digest)
    joined = "".join(name + "\0" + row["sha256"] + "\n" for name, row in sorted(actual.items()))
    identity = hashlib.sha256(joined.encode()).hexdigest()
    if (len(expected) != len(rows) or expected != actual or len(actual) != base.PROJECT_COUNT
            or sum(row["size_bytes"] for row in actual.values()) != base.PROJECT_BYTES
            or manifest["file_count"] != base.PROJECT_COUNT or manifest["total_bytes"] != base.PROJECT_BYTES
            or identity != base.PROJECT_SHA256 or manifest["snapshot_sha256"] != identity
            or manifest["source_git_head"] != "e0e7b74ff2a69083f952c1ad436bc92d4fe1ec6d"
            or "AGENTS.md" not in actual):
        raise ValueError("exact reviewed project snapshot required")
    sanitized = {"schema": 1, "revision": manifest["source_git_head"], "files": actual,
                 "snapshot_sha256": identity,
                 "limitations": "Exported ordinary source subset; not a standalone build. Request exact missing ordinary dependencies."}
    return profile, sanitized


def verify_context(directory, profile, digest):
    if (profile["source_manifest_sha256"] != SOURCE_MANIFEST_SHA256
            or profile["task_sha256"] not in TASK_SHA256S
            or profile["context_sha256"] != CONTEXT_SHA256
            or digest(directory / "support/application-context.md") != CONTEXT_SHA256):
        raise ValueError("neutral candidate/task/context identity changed")


def verify_staged_inputs(directory, profile, manifest, identity, safe, digest):
    """Check copied inputs against admitted identities before any copied import."""
    source = safe(directory / "source")
    actual, directories = {}, set()
    for path in source.rglob("*"):
        safe(path)
        if path.is_file():
            actual[str(path.relative_to(source))] = digest(path)
        elif path.is_dir():
            directories.add(str(path.relative_to(source)))
        else:
            raise ValueError("nonregular neutral staged source")
    if (actual != manifest["copied_files_sha256"]
            or directories != base.source_directories(manifest["copied_files_sha256"])):
        raise ValueError("neutral staged source differs from admitted manifest")
    if base.inventory(safe(directory / "fixture"), safe, digest) != identity["files"]:
        raise ValueError("neutral staged fixture differs from admitted identity")
    if digest(directory / "task.txt") != profile["task_sha256"]:
        raise ValueError("neutral staged task differs from admitted identity")
    verify_context(directory, profile, digest)


def context_input(directory):
    return {"type": "text", "text": (directory / "support/application-context.md").read_bytes().decode("utf-8")}


def stage(directory, profile, identity, children, write_json):
    if type(children) is not int or children != 11:
        raise ValueError("neutral route requires the fixed available child capacity")
    for name in ("work/actors", "work/outbox", "work/accounting/web",
                 "relay/inbox", "relay/dependencies", "relay/control", "support"):
        (directory / name).mkdir(parents=True, mode=0o700, exist_ok=True)
    for name in ACTOR_SUPPORT:
        shutil.copyfile(Path(__file__).with_name(name), directory / "support" / name)
    shutil.copyfile(profile["context"], directory / "support/application-context.md")
    (directory / "relay/inbox/.service.lock").touch(mode=0o400, exist_ok=False)
    write_json(directory / "support/profile.json", {"name": NAME, "limits": LIMITS,
               "relay_services": profile["relay_services"], "max_open_children": children})
    write_json(directory / "support/project-source.json", identity)
    allocation = {"total_cap": LIMITS["web_items"], "runs": {"author": LIMITS["web_items"]}}
    write_json(directory / "support/web-allocations.json", allocation)
    write_json(directory / "work/accounting/web/allocations.json", allocation)
    note = f"""# Execution environment

is at {directory / 'support/application-context.md'} and is supplied unchanged.
